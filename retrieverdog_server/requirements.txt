# ============================================================
# RetrieverDog 后端依赖清单
# 安装命令：pip install -r requirements.txt
# ============================================================

# ---------- Web 框架 ----------
# FastAPI：高性能 Python Web 框架，用于构建 API 接口
fastapi
# Uvicorn：ASGI 服务器，运行 FastAPI 应用
uvicorn
# 处理 multipart/form-data 文件上传
python-multipart
# 加载 .env 环境变量文件
python-dotenv
# 数据校验（FastAPI 默认使用，显式列出方便管理版本）
pydantic

# ---------- LlamaIndex 核心 ----------
# LlamaIndex 主框架，提供 RAG、索引、检索等全套能力
llama-index
# OpenAI 兼容格式的 LLM 封装，用于接入千问等兼容 OpenAI API 的模型
llama-index-llms-openai-like
# 阿里云 DashScope 嵌入模型（千问文本向量）
llama-index-embeddings-dashscope
# ChromaDB 向量数据库的 LlamaIndex 适配层
llama-index-vector-stores-chroma
# BM25 关键词检索器（稀疏检索，与向量检索互补）
llama-index-retrievers-bm25
# 阿里云 DashScope 云端重排序（qwen-rerank / gte-rerank）
llama-index-postprocessor-dashscope-rerank
# LlamaParse：官方 PDF 解析服务，解析效果远好于本地工具
llama-parse

# ---------- 文档解析 ----------
# Word 文档（.docx）文本提取
python-docx
# Excel 读写
openpyxl
# PPT 文档（.pptx）文本提取
python-pptx
# 数据分析库，用于读取 Excel/CSV 并转为文本
pandas

# ---------- 图片 OCR ----------
# Tesseract OCR 的 Python 封装（需要本机安装 Tesseract-OCR）
pytesseract
# Python 图像处理库（OCR 读取图片）
Pillow
# PyMuPDF（fitz）：PDF 转图片，用于 OCR 备用方案
PyMuPDF

# ---------- 向量数据库 ----------
# ChromaDB：轻量级本地向量数据库，支持持久化存储
chromadb
